<script>


const phases = [
    [
        {a:5, b:2, op:"-"},
        {a:3, b:1, op:"-"},
        {a:4, b:2, op:"+"}
    ],


    [
        {a:8, b:3, op:"-"},
        {a:10, b:4, op:"-"},
        {a:7, b:5, op:"+"}
    ],


    [
        {a:12, b:5, op:"-"},
        {a:15, b:7, op:"-"},
        {a:9, b:6, op:"+"}
    ],


    [
        {a:18, b:9, op:"-"},
        {a:20, b:8, op:"-"},
        {a:14, b:7, op:"+"}
    ],


    [
        {a:25, b:12, op:"-"},
        {a:30, b:15, op:"-"},
        {a:18, b:9, op:"+"}
    ]
];


let currentPhase = 0;
let currentQuestion = 0;
let score = 0;
let attempts = 2;


function startGame(){
    document.getElementById("startScreen").style.display = "none";
    document.getElementById("gameScreen").style.display = "flex";
    loadQuestion();
}


function loadQuestion(){


    document.getElementById("nextBtn").style.display = "none";
    document.getElementById("message").innerHTML = "";
    document.getElementById("answer").value = "";


    attempts = 2;
    updateAttempts();


    const q = phases[currentPhase][currentQuestion];


    const colorClass = (currentQuestion % 2 === 0) ? "pink" : "blue";


    document.getElementById("question").className = "question " + colorClass;


    document.getElementById("question").innerHTML =
        `${q.a} ${q.op} ${q.b} = ?`;


    document.getElementById("phaseText").innerHTML =
        `Fase ${currentPhase + 1}`;


    createApples(q);
}


function createApples(q){


    const applesDiv = document.getElementById("apples");
    applesDiv.innerHTML = "";


    let total = q.a;


    for(let i=0; i<total; i++){


        const apple = document.createElement("div");
        apple.classList.add("apple");


        if(q.op === "-" && i >= (q.a - q.b)){
            apple.innerHTML = "🍎";
            apple.style.opacity = "0.3";
        }else{
            apple.innerHTML = "🍎";
        }


        applesDiv.appendChild(apple);
    }
}


function checkAnswer(){


    const q = phases[currentPhase][currentQuestion];


    let correct;


    if(q.op === "+"){
        correct = q.a + q.b;
    }else{
        correct = q.a - q.b;
    }


    const userAnswer =
        Number(document.getElementById("answer").value);


    const msg = document.getElementById("message");


    if(userAnswer === correct){


        score += 5;


        document.getElementById("score").innerHTML =
            `⭐ Pontos: ${score}`;


        msg.innerHTML =
            `<div class="success">
                🎉 Parabéns! Você acertou!
            </div>`;


        createConfetti();


        document.getElementById("nextBtn").style.display = "inline-block";


    }else{


        attempts--;
        updateAttempts();


        if(attempts > 0){


            msg.innerHTML =
            `<div class="error">
                ❌ Resposta incorreta.<br><br>
                Observe a conta com atenção.<br>
                ${q.a} ${q.op} ${q.b}
            </div>`;


        }else{


            let explanation = "";


            if(q.op === "-"){
                explanation =
                `Para resolver ${q.a} - ${q.b},
                retiramos ${q.b} maçãs de ${q.a}.<br>
                Resultado correto: <strong>${correct}</strong>`;
            }else{
                explanation =
                `Para resolver ${q.a} + ${q.b},
                juntamos as quantidades.<br>
                Resultado correto: <strong>${correct}</strong>`;
            }


            msg.innerHTML =
            `<div class="error">
                ❌ Você usou as duas tentativas.<br><br>
                ${explanation}
            </div>`;


            document.getElementById("nextBtn").style.display =
                "inline-block";
        }
    }
}


function nextQuestion(){


    currentQuestion++;


    if(currentQuestion >= 3){
        currentQuestion = 0;
        currentPhase++;
    }


    if(currentPhase >= 5){


        document.getElementById("gameScreen").style.display = "none";
        document.getElementById("finalScreen").style.display = "flex";


        document.getElementById("finalScore").innerHTML =
            `Pontuação Final: ${score} pontos`;


        createConfetti(120);


        return;
    }


    loadQuestion();
}


function updateAttempts(){
    document.getElementById("attempts").innerHTML =
        `Tentativas: ${attempts}`;
}


function createConfetti(quantity = 50){


    const colors = [
        "#ff66c4",
        "#7d5cff",
        "#66b3ff",
        "#ffffff"
    ];


    for(let i=0; i<quantity; i++){


        const confetti = document.createElement("div");


        confetti.classList.add("confetti");


        confetti.style.left = Math.random() * 100 + "vw";


        confetti.style.background =
            colors[Math.floor(Math.random()*colors.length)];


        confetti.style.animationDuration =
            (Math.random()*3 + 2) + "s";


        confetti.style.width =
            (Math.random()*10 + 6) + "px";


        confetti.style.height =
            confetti.style.width;


        document.body.appendChild(confetti);


        setTimeout(()=>{
            confetti.remove();
        },5000);
    }
}
